function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var students = JSON.parse(props.getProperty('students') || '[]');
  var results = JSON.parse(props.getProperty('results') || '[]');
  var revokedIds = JSON.parse(props.getProperty('revokedIds') || '[]');
  return ContentService.createTextOutput(JSON.stringify({ students: students, results: results, revokedIds: revokedIds }))
    .setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var props = PropertiesService.getScriptProperties();
    var body = JSON.parse(e.postData.contents || '{}');
    var curRevoked = JSON.parse(props.getProperty('revokedIds') || '[]');
    if (Array.isArray(body.revokedIds)) {
      body.revokedIds.forEach(function(id) { if (curRevoked.indexOf(id) === -1) curRevoked.push(id); });
      props.setProperty('revokedIds', JSON.stringify(curRevoked));
    }
    if (Array.isArray(body.students)) {
      var curStudents = JSON.parse(props.getProperty('students') || '[]');
      var sMap = {};
      curStudents.concat(body.students).forEach(function(s) {
        if (s && s.loginId && curRevoked.indexOf(s.loginId) === -1) sMap[s.loginId] = s;
      });
      props.setProperty('students', JSON.stringify(Object.keys(sMap).map(function(k) { return sMap[k]; })));
    }
    if (Array.isArray(body.results)) {
      var curResults = JSON.parse(props.getProperty('results') || '[]');
      var rMap = {};
      curResults.concat(body.results).forEach(function(r) {
        if (r && r.resultId) rMap[r.resultId] = r;
      });
      props.setProperty('results', JSON.stringify(Object.keys(rMap).map(function(k) { return rMap[k]; })));
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
